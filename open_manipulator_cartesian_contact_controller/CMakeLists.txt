cmake_minimum_required(VERSION 3.16)
project(open_manipulator_cartesian_contact_controller LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(open_manipulator_friction_diagnostic REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(osqp_vendor REQUIRED)
find_package(osqp REQUIRED CONFIG)

add_library(cartesian_law SHARED src/cartesian_law.cpp)
target_compile_features(cartesian_law PUBLIC cxx_std_17)
target_compile_options(cartesian_law PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(cartesian_law PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(cartesian_law PUBLIC Eigen3::Eigen PRIVATE osqp::osqp)

add_library(cartesian_contact_controller SHARED src/cartesian_controller.cpp)
target_compile_features(cartesian_contact_controller PUBLIC cxx_std_17)
target_compile_options(cartesian_contact_controller PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(cartesian_contact_controller PUBLIC rcl_interfaces geometry_msgs)
target_link_libraries(cartesian_contact_controller PUBLIC cartesian_law
  open_manipulator_friction_diagnostic::friction_compensation_controller)
pluginlib_export_plugin_description_file(controller_interface cartesian_contact_plugin.xml)
install(TARGETS cartesian_law cartesian_contact_controller LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/l100_cartesian_leader.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/cartesian_leader_math.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
